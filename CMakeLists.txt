cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" GREEK_COAST_VERSION LIMIT_COUNT 1)
project(GreekCoast VERSION ${GREEK_COAST_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(GNUInstallDirs)
option(GREEK_COAST_STANDALONE_SETTINGS "Build optional standalone Qt settings" ON)
option(GREEK_COAST_SETTINGS_BRIDGE "Build optional per-user GTK settings compatibility bridge" OFF)
set(XSCREENSAVER_HACK_DIR "${CMAKE_INSTALL_LIBEXECDIR}/xscreensaver" CACHE STRING "XScreenSaver executable installation directory")
set(XSCREENSAVER_CONFIG_DIR "${CMAKE_INSTALL_DATADIR}/xscreensaver/config" CACHE STRING "XScreenSaver configurator XML installation directory")
find_package(PkgConfig REQUIRED)
pkg_check_modules(RENDER REQUIRED IMPORTED_TARGET epoxy x11 egl libpng)
find_package(Threads REQUIRED)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui)
if(GREEK_COAST_STANDALONE_SETTINGS)
  find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets)
  add_executable(greek-coast-settings src/settings.cpp)
  target_link_libraries(greek-coast-settings PRIVATE environment Qt6::Widgets)
  install(TARGETS greek-coast-settings RUNTIME DESTINATION "${XSCREENSAVER_HACK_DIR}")
endif()
add_library(environment STATIC src/environment.cpp)
target_include_directories(environment PUBLIC src)
target_link_libraries(environment PUBLIC Qt6::Core)
add_library(ocean STATIC src/ocean.cpp)
target_include_directories(ocean PUBLIC src PRIVATE vendor/pocketfft)
target_compile_definitions(ocean PRIVATE POCKETFFT_CACHE_SIZE=8 POCKETFFT_NO_MULTITHREADING)
add_library(road STATIC src/road.cpp)
add_library(people STATIC src/people.cpp)
add_library(fish STATIC src/fish.cpp)
target_include_directories(fish PUBLIC src)
foreach(target road people)
  target_include_directories(${target} PUBLIC src)
endforeach()
add_library(boat-model STATIC src/boat_model.cpp)
target_include_directories(boat-model PUBLIC src)
target_link_libraries(boat-model PUBLIC ocean Qt6::Gui)
add_executable(greek-coast src/main.cpp src/renderer.cpp src/boat_renderer.cpp)
target_link_libraries(greek-coast PRIVATE ocean road people fish environment boat-model PkgConfig::RENDER Threads::Threads)
foreach(target greek-coast ocean environment road people boat-model)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
get_filename_component(hack_full "${XSCREENSAVER_HACK_DIR}" ABSOLUTE BASE_DIR "${CMAKE_INSTALL_PREFIX}")
get_filename_component(data_full "${CMAKE_INSTALL_DATADIR}/greek-coast" ABSOLUTE BASE_DIR "${CMAKE_INSTALL_PREFIX}")
get_filename_component(bin_full "${CMAKE_INSTALL_BINDIR}" ABSOLUTE BASE_DIR "${CMAKE_INSTALL_PREFIX}")
get_filename_component(bridge_full "${CMAKE_INSTALL_LIBDIR}/greek-coast" ABSOLUTE BASE_DIR "${CMAKE_INSTALL_PREFIX}")
file(RELATIVE_PATH GREEK_COAST_DATA_RELATIVE "${hack_full}" "${data_full}")
file(RELATIVE_PATH GREEK_COAST_BINARY_FROM_BINDIR "${bin_full}" "${hack_full}/greek-coast")
file(RELATIVE_PATH GREEK_COAST_BRIDGE_FROM_BINDIR "${bin_full}" "${bridge_full}/libgreek-coast-xs-config.so")
file(RELATIVE_PATH GREEK_COAST_BRIDGE_DATA_RELATIVE "${bridge_full}" "${data_full}")
file(RELATIVE_PATH GREEK_COAST_DATA_FROM_BINDIR "${bin_full}" "${data_full}")
target_compile_definitions(greek-coast PRIVATE GREEK_COAST_DATA_RELATIVE="${GREEK_COAST_DATA_RELATIVE}")
# Build-tree execution uses the same data layout as installed execution.
file(COPY assets shaders DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/share/greek-coast")
configure_file(greek-coast.xml "${CMAKE_CURRENT_BINARY_DIR}/share/greek-coast/greek-coast.xml" COPYONLY)
if(GREEK_COAST_SETTINGS_BRIDGE)
  pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0)
  add_library(greek-coast-xs-config SHARED src/xscreensaver_config.cpp src/settings_scroll.cpp)
  target_compile_definitions(greek-coast-xs-config PRIVATE GREEK_COAST_BRIDGE_DATA_RELATIVE="${GREEK_COAST_BRIDGE_DATA_RELATIVE}")
  target_link_libraries(greek-coast-xs-config PRIVATE ${CMAKE_DL_LIBS} PkgConfig::GTK3)
  configure_file(scripts/xscreensaver-settings.sh.in greek-coast-xscreensaver-settings @ONLY FILE_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
  install(TARGETS greek-coast-xs-config LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}/greek-coast")
  install(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/greek-coast-xscreensaver-settings" DESTINATION "${CMAKE_INSTALL_BINDIR}")
endif()
configure_file(scripts/register.py.in greek-coast-register @ONLY FILE_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
install(TARGETS greek-coast RUNTIME DESTINATION "${XSCREENSAVER_HACK_DIR}")
install(FILES greek-coast.xml DESTINATION "${XSCREENSAVER_CONFIG_DIR}")
install(FILES greek-coast.xml DESTINATION "${XSCREENSAVER_CONFIG_DIR}" RENAME greek_coast.xml)
install(FILES greek-coast.xml DESTINATION "${CMAKE_INSTALL_DATADIR}/greek-coast")
install(DIRECTORY assets shaders DESTINATION "${CMAKE_INSTALL_DATADIR}/greek-coast" FILES_MATCHING PATTERN "*.png" PATTERN "*.frag" PATTERN "*.vert")
install(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/greek-coast-register" DESTINATION "${CMAKE_INSTALL_BINDIR}")
install(FILES LICENSE THIRD_PARTY_NOTICES.md vendor/pocketfft/LICENSE.md DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/greek-coast")
install(DIRECTORY LICENSES/ DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/greek-coast")
configure_file(cmake/uninstall.cmake.in uninstall.cmake @ONLY)
add_custom_target(uninstall COMMAND ${CMAKE_COMMAND} -P "${CMAKE_CURRENT_BINARY_DIR}/uninstall.cmake")
include(CTest)

  add_executable(ocean-test tests/ocean_test.cpp)
  target_link_libraries(ocean-test PRIVATE ocean)
  add_test(NAME spectral-ocean COMMAND ocean-test)
  add_executable(boat-test tests/boat_test.cpp)
  target_link_libraries(boat-test PRIVATE boat-model)
  add_test(NAME fishing-boat-model COMMAND boat-test)
  add_executable(fish-test tests/fish_test.cpp)
  target_link_libraries(fish-test PRIVATE fish)
  add_test(NAME fish-flocking COMMAND fish-test)
  add_executable(people-test tests/people_test.cpp)
  target_link_libraries(people-test PRIVATE people)
  add_test(NAME people-dynamics COMMAND people-test)
  add_executable(road-test tests/road_test.cpp)
  target_link_libraries(road-test PRIVATE road)
  add_test(NAME village-road-routing COMMAND road-test)
  add_executable(preview-reentry-test tests/preview_reentry_test.cpp)

  target_link_libraries(termination-test PRIVATE Threads::Threads)
  add_test(NAME embedded-preview-termination COMMAND termination-test)

  if(GREEK_COAST_SETTINGS_BRIDGE)
  add_executable(settings-scroll-test tests/settings_scroll_test.cpp)

  set_tests_properties(native-settings-lookup PROPERTIES ENVIRONMENT "LD_PRELOAD=$<TARGET_FILE:greek-coast-xs-config>")
  endif()
  add_executable(environment-test tests/environment_test.cpp)
  target_link_libraries(environment-test PRIVATE environment)
  add_test(NAME sun-and-cycle COMMAND environment-test)
  if(GREEK_COAST_STANDALONE_SETTINGS)
  add_test(NAME settings-window COMMAND greek-coast-settings --self-test)
  set_tests_properties(settings-window PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;XDG_CONFIG_HOME=${CMAKE_CURRENT_BINARY_DIR}/test-config" TIMEOUT 20)
  endif()
  add_test(NAME opengl-render COMMAND greek-coast --headless --width 640 --height 360 --time 9 --snapshot "${CMAKE_CURRENT_BINARY_DIR}/test-frame.png")
  set_tests_properties(opengl-render PROPERTIES ENVIRONMENT "LIBGL_ALWAYS_SOFTWARE=1;XDG_CONFIG_HOME=${CMAKE_CURRENT_BINARY_DIR}/test-config" TIMEOUT 30)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME native-settings-contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/settings_contract_test.py ${CMAKE_CURRENT_SOURCE_DIR}/greek-coast.xml $<TARGET_FILE:greek-coast>)
    add_test(NAME portable-install COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/install_test.py ${CMAKE_CURRENT_BINARY_DIR})
    set_tests_properties(portable-install PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
    add_test(NAME user-registration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/registration_test.py ${CMAKE_CURRENT_BINARY_DIR}/greek-coast-register $<TARGET_FILE:greek-coast>)

    if(PIL_RESULT EQUAL 0)
      add_test(NAME fish-images COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/fish_images_test.py $<TARGET_FILE:greek-coast>)
